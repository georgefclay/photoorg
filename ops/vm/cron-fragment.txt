# Cron fragment installed in `ubuntu`'s crontab on the VM.
#
# The crontab is SHARED with several other sites (crafttags, mineralsmonitor,
# agenticcomplete, etc). NEVER install with `crontab cron-fragment.txt` —
# that would replace the whole crontab and delete the other apps' lines.
#
# Replace only the photoorg block, with ANCHORED markers: these comment
# lines contain the marker text, so an unanchored pattern matches here too
# and drags this header into the crontab (found in Phase 12). Back up first:
#   crontab -l > ~/backups/crontab-$(date +%F).txt
#   ( crontab -l | sed '/^# --- photoorg ---$/,/^# --- END photoorg ---$/d' ; \
#     sed -n '/^# --- photoorg ---$/,/^# --- END photoorg ---$/p' ops/vm/cron-fragment.txt ) | crontab -
#
# Times are in the system's local TZ (America/Chicago).

# --- photoorg ---
# Nightly Postgres dump at 03:20 local (America/Chicago). 14-day retention.
20 3 * * *  /home/ubuntu/photoorg/ops/pg_dump_photoorg.sh >> /var/log/photoorg-cron.log 2>&1
# Phase 12: hourly download sweep — deletes zips past their 72 hours (the

# whose worker is gone. Silent unless it did something.
17 * * * *  cd /home/ubuntu/photoorg/web && /usr/local/bin/node tools/download-sweep.js >> /var/log/photoorg-cron.log 2>&1
